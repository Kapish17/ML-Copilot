# Dependencies of the language-model layer.
#
# Install alongside the other components for a full development environment:
#
#   pip install -r backend/requirements.txt -r ml/requirements.txt \
#               -r rag/requirements.txt -r llm/requirements.txt

# The official OpenAI SDK, used through its chat-completions API. That API
# rather than a vendor-specific one is deliberate: the same provider
# implementation talks to OpenAI, Azure OpenAI, vLLM, Ollama, LM Studio,
# OpenRouter and anything else that speaks the shape, by pointing LLM_BASE_URL
# at it. One provider therefore covers hosted models and a model running on
# the developer's own machine.
#
# Nothing is imported at package-import time and no client is built until a
# generation is actually requested, so the layer imports — and the whole test
# suite runs — with this installed but no API key configured.
openai==3.6.0
